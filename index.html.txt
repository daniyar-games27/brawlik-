 <div class="box">
  <h1>RIFT RUMBLE</h1>
  <h2>BEATBOUND ARENA</h2>
  <p>3v3 · Pulse Clash · Crown control</p>
  <p>Захвати центральную корону и удерживай её!</p>
  <p><small>WASD — движение · мышь — прицел · ЛКМ — атака<br>
  SPACE — рывок · E — особый удар · 1/2/3 — карты</small></p>
  <button onclick="start()">PLAY</button>
 </div>

<script>
const canvas=document.querySelector("canvas");
const ctx=canvas.getContext("2d");
let W,H;
function resize(){W=canvas.width=innerWidth;H=canvas.height=innerHeight}
resize();onresize=resize;

const keys={};
onkeydown=e=>{
 keys[e.key.toLowerCase()]=true;
 if(e.key===" "){e.preventDefault();dash()}
 if(e.key.toLowerCase()==="e")special();
 if(["1","2","3"].includes(e.key))useCard(+e.key);
 if(e.key==="Escape")paused=!paused;
};
onkeyup=e=>keys[e.key.toLowerCase()]=false;

let mouse={x:W/2,y:H/2};
canvas.onmousemove=e=>{mouse.x=e.clientX;mouse.y=e.clientY};
canvas.onclick=()=>attack();

let players=[],bullets=[],effects=[];
let running=false,paused=false;
let blue=0,red=0,time=120,energy=5;
let dashCD=0,specialCD=0,attackCD=0;
let rhythm=0, pulseWarning=false;
let crown={x:0,y:0,r:65};
let player;

function start(){
 document.getElementById("menu").style.display="none";
 blue=0;red=0;time=120;energy=5;
 players=[];bullets=[];effects=[];
 player=createPlayer(W/2-80,H/2,"blue",true,"Pebblepop");
 players.push(player);
 for(let i=0;i<2;i++)players.push(createPlayer(W/2-180,H/2+i*70,"blue",false,"Ally"));
 for(let i=0;i<3;i++)players.push(createPlayer(W/2+120+i*60,H/2+i*80,"red",false,"Rival"));
 running=true;paused=false;
}

function createPlayer(x,y,team,human,name){
 return {x,y,team,human,name,r:16,hp:100,
 vx:0,vy:0,attack:0,cd:0,dash:0,shield:0,
 speed:human?3.5:2.1,alive:true,respawn:0};
}

function distance(a,b){return Math.hypot(a.x-b.x,a.y-b.y)}
function clamp(v,a,b){return Math.max(a,Math.min(b,v))}

function attack(){
 if(!running||paused||!player||attackCD>0)return;
 attackCD=.28;
 let angle=Math.atan2(mouse.y-player.y,mouse.x-player.x);
 bullets.push({x:player.x,y:player.y,vx:Math.cos(angle)*8,
 vy:Math.sin(angle)*8,team:"blue",life:40,damage:18});
}

function dash(){
 if(!running||paused||dashCD>0)return;
 let dx=(keys.d?1:0)-(keys.a?1:0);
 let dy=(keys.s?1:0)-(keys.w?1:0);
 if(!dx&&!dy){dx=mouse.x-player.x;dy=mouse.y-player.y}
 let l=Math.hypot(dx,dy)||1;
 player.vx+=dx/l*10;player.vy+=dy/l*10;
 dashCD=1.5;
 effects.push({x:player.x,y:player.y,life:15,color:"#71f5ff"});
}

function special(){
 if(!running||paused||specialCD>0||energy<2)return;
 energy-=2;specialCD=4;
 effects.push({x:player.x,y:player.y,life:40,color:"#b78cff",ring:true});
 for(let p of players){
  if(p.team==="red"&&p.alive&&distance(p,player)<140){
   p.hp-=35;
   if(p.hp<=0)down(p);
  }
 }
}

function useCard(n){
 if(!running||paused||energy<1)return;
 energy--;
 if(n===1){
  effects.push({x:player.x,y:player.y,life:25,color:"#ffd56a"});
  for(let p of players)if(p.team==="red"&&distance(p,player)<110){p.hp-=25;if(p.hp<=0)down(p)}
 }
 if(n===2){
  player.shield=120;
  effects.push({x:player.x,y:player.y,life:30,color:"#79ffb5"});
 }
 if(n===3){
  player.hp=Math.min(100,player.hp+35);
  effects.push({x:player.x,y:player.y,life:30,color:"#91ff8b"});
 }
}

function down(p){
 p.alive=false;p.respawn=3;
 if(p.team==="blue")red++;
 else blue++;
}

function respawn(p){
 p.hp=100;p.alive=true;
 p.x=p.team==="blue"?W/2-100:W/2+100;
 p.y=H/2+Math.random()*100-50;
}

function update(){
 if(!running||paused)return;
 time-=1/60;
 if(time<=0||blue>=5||red>=5){endGame();return}
 energy=Math.min(5,energy+1/240);
 dashCD=Math.max(0,dashCD-1/60);
 specialCD=Math.max(0,specialCD-1/60);
 attackCD=Math.max(0,attackCD-1/60);
 rhythm+=1/60;
 pulseWarning=(rhythm%8)>5.5&&(rhythm%8)<6.7;

 // Player movement
 let dx=(keys.d?1:0)-(keys.a?1:0);
 let dy=(keys.s?1:0)-(keys.w?1:0);
 let l=Math.hypot(dx,dy)||1;
 player.vx+=(dx/l)*player.speed*.25;
 player.vy+=(dy/l)*player.speed*.25;
 player.vx*=.83;player.vy*=.83;
 player.x+=player.vx;player.y+=player.vy;
 player.x=clamp(player.x,30,W-30);player.y=clamp(player.y,80,H-50);

 // Bot AI
 for(let p of players){
  if(p.human)continue;
  if(!p.alive){p.respawn-=1/60;if(p.respawn<=0)respawn(p);continue}
  let target=player;
  let nearest=Infinity;
  for(let q of players){
   if(q.team!==p.team&&q.alive&&distance(p,q)<nearest){target=q;nearest=distance(p,q)}
  }
  let cx=W/2,cy=H/2;
  let tx=nearest<180?target.x:cx;
  let ty=nearest<180?target.y:cy;
  let d=Math.hypot(tx-p.x,ty-p.y)||1;
  p.x+=((tx-p.x)/d)*p.speed;
  p.y+=((ty-p.y)/d)*p.speed;
  if(nearest<150&&Math.random()<.025){
   bullets.push({x:p.x,y:p.y,vx:(target.x-p.x)/d*5,
    vy:(target.y-p.y)/d*5,team:p.team,life:40,damage:12});
  }
  p.x=clamp(p.x,30,W-30);p.y=clamp(p.y,80,H-50);
 }

 // Objective: crown control
 let b=players.filter(p=>p.alive&&p.team==="blue"&&distance(p,crown)<crown.r).length;
 let r=players.filter(p=>p.alive&&p.team==="red"&&distance(p,crown)<crown.r).length;
 if(b>r&&b>0)blue+=1/300;
 if(r>b&&r>0)red+=1/300;

 // Rhythm pulse
 if(rhythm%8>6.7&&rhythm%8<6.75){
  for(let p of players){
   if(p.alive&&distance(p,crown)<crown.r+15){
    if(p.shield>0)p.shield-=20;
    else {p.hp-=15;if(p.hp<=0)down(p)}
   }
  }
  effects.push({x:crown.x,y:crown.y,life:35,color:"#ff73e8",ring:true});
 }

 // Bullets
 for(let b of bullets){
  b.x+=b.vx;b.y+=b.vy;b.life--;
  for(let p of players){
   if(!p.alive||p.team===b.team)continue;
   if(Math.hypot(p.x-b.x,p.y-b.y)<p.r+5){
    p.hp-=b.damage;
    b.life=0;
    if(p.hp<=0)down(p);
   }
  }
 }
 bullets=bullets.filter(b=>b.life>0);

 effects.forEach(e=>e.life--);
 effects=effects.filter(e=>e.life>0);

 players.forEach(p=>p.shield=Math.max(0,p.shield-1));
}

function draw(){
 ctx.clearRect(0,0,W,H);
 ctx.fillStyle="#090d1b";ctx.fillRect(0,0,W,H);

 // Grid
 ctx.strokeStyle="#19243b";ctx.lineWidth=1;
 for(let x=0;x<W;x+=40){ctx.beginPath();ctx.moveTo(x,0);ctx.lineTo(x,H);ctx.stroke()}
 for(let y=0;y<H;y+=40){ctx.beginPath();ctx.moveTo(0,y);ctx.lineTo(W,y);ctx.stroke()}

 // Crown zone
 crown.x=W/2;crown.y=H/2;
 ctx.beginPath();ctx.arc(crown.x,crown.y,crown.r,0,Math.PI*2);
 ctx.fillStyle="#30214c";ctx.fill();
 ctx.strokeStyle=pulseWarning?"#ff69eb":"#bd87ff";
 ctx.lineWidth=3;ctx.stroke();
 ctx.fillStyle="#ffe48c";ctx.font="32px Arial";
 ctx.textAlign="center";ctx.fillText("♛",crown.x,crown.y+10);
 ctx.font="12px Arial";ctx.fillText("CROWN",crown.x,crown.y+35);

 // Bullets
 for(let b of bullets){
  ctx.fillStyle=b.team==="blue"?"#79f8ff":"#ff7979";
  ctx.beginPath();ctx.arc(b.x,b.y,5,0,Math.PI*2);ctx.fill();
 }

 // Characters
 for(let p of players){
  if(!p.alive)continue;
  ctx.fillStyle=p.team==="blue"?"#45dcff":"#ff5b75";
  ctx.beginPath();ctx.arc(p.x,p.y,p.r,0,Math.PI*2);ctx.fill();
  ctx.strokeStyle="white";ctx.lineWidth=2;ctx.stroke();
  if(p.shield>0){
   ctx.beginPath();ctx.arc(p.x,p.y,p.r+6,0,Math.PI*2);
   ctx.strokeStyle="#a5ffbd";ctx.lineWidth=3;ctx.stroke();
  }
  ctx.fillStyle="#fff";
  ctx.fillRect(p.x-18,p.y-28,36,5);
  ctx.fillStyle="#6aff89";
  ctx.fillRect(p.x-18,p.y-28,36*(p.hp/100),5);
  ctx.font="11px Arial";ctx.textAlign="center";
  ctx.fillStyle="white";ctx.fillText(p.name,p.x,p.y-34);
 }

 // Effects
 for(let e of effects){
  ctx.globalAlpha=e.life/40;
  ctx.strokeStyle=e.color;ctx.lineWidth=4;
  ctx.beginPath();ctx.arc(e.x,e.y,50+(40-e.life)*3,0,Math.PI*2);ctx.stroke();
  ctx.globalAlpha=1;
 }

 if(pulseWarning){
  ctx.fillStyle="#ff60df22";ctx.fillRect(0,0,W,H);
  ctx.fillStyle="#ffb2f4";ctx.font="bold 24px Arial";
  ctx.textAlign="center";ctx.fillText("PULSE INCOMING!",W/2,90);
 }

 // HUD
 document.getElementById("score").textContent=
 "BLUE "+Math.floor(blue)+" — "+Math.floor(red)+" RED";
 document.getElementById("timer").textContent=Math.ceil(time);
 document.getElementById("hp").textContent=Math.max(0,Math.ceil(player.hp));
 document.getElementById("energy").textContent=energy.toFixed(1);
}

function endGame(){
 running=false;
 document.getElementById("menu").style.display="flex";
 document.querySelector("#menu .box").innerHTML=
 `<h1>${blue>red?"BLUE VICTORY!":"RED VICTORY!"}</h1>
 <p>Счёт: ${Math.floor(blue)} — ${Math.floor(red)}</p>
 <button onclick="start()">PLAY AGAIN</button>`;
}

function loop(){
 update();draw();requestAnimationFrame(loop);
}
loop();
</script>
